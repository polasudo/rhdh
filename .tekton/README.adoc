To switch from a rhdh-1- build to a rhdh-1-y- build, apply the following transformations:

```
# rename from -1- to -1-y- (eg., -1-4-)
y=4
for d in *-1-p*; do git mv $d ${d/1-p/1-${y}-p}; done
# replace strings
for d in *1-${y}-p*; do echo $d; 
    sed -i $d -r \
        -e "s/-1-on/-1-${y}-on/g" \
        -e "s/-1-p/-1-${y}-p/g" \
        -e "s/rhdh-1$/rhdh-1-${y}/g" \
        -e "s/rhdh-hub-1$/rhdh-hub-1-${y}/g" \
        -e "s/rhdh-operator-1$/rhdh-operator-1-${y}/g" \
        -e "s/rhdh-operator-bundle-1$/rhdh-operator-bundle-1-${y}/g"; \
done
```
